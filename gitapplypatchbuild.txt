+

+    fun string(
+        key: String,
+        moduleName: String,
+        explicit: String?,
+    ): String =
+        explicit
+            ?: find("module.$moduleName.$key")
+            ?: find(key)
+            ?: FALLBACKS.getValue(key)
+
+    fun int(
+        key: String,
+        moduleName: String,
+        explicit: Int?,
+    ): Int = string(key = key, moduleName = moduleName, explicit = explicit?.toString()).toInt()
+
+    private fun find(key: String): String? = values.getProperty(key)?.trim()?.takeIf { it.isNotEmpty() }
+}
+
+private fun Project.loadProjectConfig(): TajsosProjectConfig {
+    val props = Properties()
+    val configFile = rootProject.file("gradle/project-config.properties")
+    if (configFile.exists()) {
+        configFile.inputStream().use(props::load)
+    }
+    return TajsosProjectConfig(props)
+}
